@Service
public class EventListenerService {

    @Autowired
    private AlbumViewRepository albumRepo;

    @Autowired
    private RecensioneViewRepository recensioneRepo;

    @Autowired
    private ConnessioneViewRepository connessioneRepo;

    @KafkaListener(topics = {
        "album-service-event-channel",
        "recensione-service-event-channel",
        "connessioni-service-event-channel"
    }, groupId = "recensioni-seguite-group")
    public void handleEvent(Object event) {
        // Album creato
        if (event instanceof AlbumCreatedEvent) {
            AlbumCreatedEvent albumEvent = (AlbumCreatedEvent) event;
            AlbumView album = new AlbumView();
            album.setId(albumEvent.getId());
            album.setTitolo(albumEvent.getTitolo());
            album.setArtista(albumEvent.getArtista());
            album.setGeneri(albumEvent.getGeneri());
            albumRepo.save(album);
        }

        // Recensione creata
        else if (event instanceof RecensioneCreatedEvent) {
            RecensioneCreatedEvent recEvent = (RecensioneCreatedEvent) event;
            RecensioneView r = new RecensioneView();
            r.setId(recEvent.getId());
            r.setAlbumId(recEvent.getAlbumId());
            r.setAutore(recEvent.getAutore());
            r.setContenuto(recEvent.getContenuto());
            recensioneRepo.save(r);
        }

        // Connessione creata
        else if (event instanceof ConnessioneCreatedEvent) {
            ConnessioneCreatedEvent connEvent = (ConnessioneCreatedEvent) event;
            ConnessioneView c = new ConnessioneView();
            c.setId(connEvent.getId());
            c.setUtente(connEvent.getUtente());
            c.setSeguito(connEvent.getSeguito());
            c.setRuolo(connEvent.getRuolo());
            connessioneRepo.save(c);
        }

        // Connessione eliminata
        else if (event instanceof ConnessioneDeletedEvent) {
            ConnessioneDeletedEvent delEvent = (ConnessioneDeletedEvent) event;
            connessioneRepo.deleteById(delEvent.getId());
        }

        // Eventi non gestiti
        else {
            System.out.println("Evento non gestito: " + event.getClass().getSimpleName());
        }
    }
}
