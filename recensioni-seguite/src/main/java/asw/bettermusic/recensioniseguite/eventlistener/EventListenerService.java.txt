@Service
public class EventListenerService {

    @Autowired
    private AlbumViewRepository albumRepo;

    @Autowired
    private RecensioneViewRepository recensioneRepo;

    @Autowired
    private ConnessioneViewRepository connessioneRepo;

    // Album creato
    @KafkaListener(topics = "album", groupId = "recensioni-seguite-group")
    public void handleAlbumCreated(AlbumCreatedEvent event) {
        AlbumView album = new AlbumView();
        album.setId(event.getId());
        album.setTitolo(event.getTitolo());
        album.setArtista(event.getArtista());
        album.setGeneri(event.getGeneri());

        albumRepo.save(album);
    }

    // Recensione creata
    @KafkaListener(topics = "recensioni", groupId = "recensioni-seguite-group")
    public void handleRecensioneCreated(RecensioneCreatedEvent event) {
        RecensioneView r = new RecensioneView();
        r.setId(event.getId());
        r.setAlbumId(event.getAlbumId());
        r.setAutore(event.getAutore());
        r.setContenuto(event.getContenuto());

        recensioneRepo.save(r);
    }

    // Connessione eliminata
    @KafkaListener(topics = "connessioni", groupId = "recensioni-seguite-group")
    public void handleConnessioneDeleted(ConnessioneDeletedEvent event) {
        connessioneRepo.deleteById(event.getId());
    }
}
